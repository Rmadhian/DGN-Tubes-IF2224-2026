    %% --------------------------------------------------------
    %% 1. LEXER & PARSER LAYER
    %% --------------------------------------------------------
    class Token {
        +TokenType type
        +string value
    }

    class Lexer {
        -string source_code
        -size_t current_pos
        -char current_char
        +tokenize() vector~Token~
    }

    class ParseTreeNode {
        +string label
        +string value
        +vector~ParseTreeNode*~ children
    }

    class Parser {
        -vector~Token~ tokens
        -int currentTokenIndex
        -bool hasError
        +parse() ParseTreeNode*
        +parseProgram() ParseTreeNode*
        +parseBlock() ParseTreeNode*
    }

    Lexer ..> Token : creates
    Parser "1" *-- "*" Token : contains
    Parser ..> ParseTreeNode : creates

    %% --------------------------------------------------------
    %% 2. AST BUILDER & AST PARSER
    %% --------------------------------------------------------
    class ASTBuilder {
        +build(ParseTreeNode* parseTreeRoot) ProgramNode*
    }
    ASTBuilder ..> ParseTreeNode : uses
    ASTBuilder ..> ProgramNode : creates

    class ASTParser {
        -string content
        -SymbolTable st
        -ProgramNode* root
        -bool error
        +parse()
        +getSymbolTable() SymbolTable
        +getASTRoot() ProgramNode*
    }

    %% --------------------------------------------------------
    %% 3. ABSTRACT SYNTAX TREE (AST) NODES
    %% --------------------------------------------------------
    class ASTNode {
        <<abstract>>
        +DataType evalType
        +int symRef
        +int lexicalLevel
        +accept(SemanticVisitor* visitor)*
    }

    class ProgramNode {
        +string name
        +vector~ASTNode*~ declarations
        +ASTNode* mainBlock
    }

    class VarDeclNode
    class ConstDeclNode
    class SubprogDeclNode
    class CompoundStmtNode
    class AssignStmtNode
    class IfStmtNode
    class WhileStmtNode
    class ForStmtNode
    class BinaryOpNode
    class UnaryOpNode
    class LiteralNode
    class VarAccessNode
    class FuncCallNode
    class WriteStatementNode

    ASTNode <|-- ProgramNode
    ASTNode <|-- VarDeclNode
    ASTNode <|-- ConstDeclNode
    ASTNode <|-- SubprogDeclNode
    ASTNode <|-- CompoundStmtNode
    ASTNode <|-- AssignStmtNode
    ASTNode <|-- IfStmtNode
    ASTNode <|-- WhileStmtNode
    ASTNode <|-- ForStmtNode
    ASTNode <|-- BinaryOpNode
    ASTNode <|-- UnaryOpNode
    ASTNode <|-- LiteralNode
    ASTNode <|-- VarAccessNode
    ASTNode <|-- FuncCallNode
    ASTNode <|-- WriteStatementNode

    ASTParser *-- ProgramNode

    %% --------------------------------------------------------
    %% 4. SYMBOL TABLE
    %% --------------------------------------------------------
    class TabEntry {
        +string identifiers
        +int link
        +ObjClass obj
        +DataType type
        +int ref
        +int nrm
        +int lev
        +int adr
    }
    class BTabEntry {
        +int blocks
        +int last
        +int lpar
        +int psze
        +int vsze
    }
    class ATabEntry {
        +int arrays
        +DataType xtyp
        +DataType etyp
        +int eref
        +int low
        +int high
        +int elsz
        +int size
    }

    class SymbolTable {
        +vector~TabEntry~ tab
        +vector~BTabEntry~ btab
        +vector~ATabEntry~ atab
        +int currentLevel
        +vector~int~ activeBlocks
        +pushScope()
        +popScope()
        +lookupTab(string name) TabEntry*
    }

    SymbolTable *-- TabEntry
    SymbolTable *-- BTabEntry
    SymbolTable *-- ATabEntry
    ASTParser *-- SymbolTable

    %% --------------------------------------------------------
    %% 5. SEMANTIC VISITOR & ANALYSIS
    %% --------------------------------------------------------
    class SemanticVisitor {
        <<abstract>>
        +SymbolTable st
        +visit(ProgramNode* node)*
        +visit(VarDeclNode* node)*
        +visit(CompoundStmtNode* node)*
    }

    class SemanticAnalyzer {
        +visit(ProgramNode* node)
    }

    class PrintASTVisitor {
        +visit(ProgramNode* node)
    }

    class ASTVisitor {
        <<alias for SemanticVisitor>>
    }

    SemanticVisitor *-- SymbolTable
    SemanticVisitor <|-- SemanticAnalyzer
    SemanticVisitor <|-- PrintASTVisitor
    SemanticVisitor <|-- ASTVisitor

    ASTNode ..> SemanticVisitor : accepts

    %% --------------------------------------------------------
    %% 6. INTERMEDIATE CODE GENERATOR (ICG)
    %% --------------------------------------------------------
    class Instruction {
        +OpCode op
        +int l
        +int a
        +int numArgs
    }

    class ICGVisitor {
        -vector~Instruction~ instructions
        -vector~string~ stringPool
        +setSymbolTable(SymbolTable symTab)
        +dispatch(ASTNode* node)
    }

    ASTVisitor <|-- ICGVisitor
    ICGVisitor "1" *-- "*" Instruction : contains

    %% --------------------------------------------------------
    %% 7. VIRTUAL MACHINE & SECURITY
    %% --------------------------------------------------------
    class VirtualMachine {
        -vector~int~ stack
        -vector~Instruction~ code
        -int IP
        -int BP
        -int SP
        +vector~string~ stringPool
        +run()
        +push(int value)
        +pop() int
        +executeOPR(int oprCode, int lField)
    }

    class Security {
        <<namespace>>
        +checkStackOverflow(int size, int add)
        +checkStackUnderflow(int size, int req)
        +checkOutOfBounds(int index, int max)
        +validateJumpTarget(int target, int max)
        +checkNumericalOverflow(long long res)
        +checkDivisionByZero(int div)
    }

    VirtualMachine "1" *-- "*" Instruction : executes
    VirtualMachine ..> Security : validates
